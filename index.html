const API_URL = "https://eb-do-axel-api.onrender.com"; const titles = { home: "Visão geral", promote: "Promover", ranks: "Hierarquia militar", history: "Histórico", audit: "Auditoria", accounts: "Contas" }; const nicknameInput = document.getElementById("nickname"); const playerName = document.getElementById("playerName"); const playerRank = document.getElementById("playerRank"); const promoteButton = document.getElementById("promoteButton"); const errorMessage = document.getElementById("errorMessage"); let jogadorEncontrado = false; let consultaAtual = 0; function clearPlayerResult() { jogadorEncontrado = false; playerName.textContent = "N/A"; playerRank.textContent = "N/A"; promoteButton.disabled = true; errorMessage.classList.add("hidden"); } function openPage(page) { document.querySelectorAll(".page").forEach(section => { section.classList.toggle("hidden", section.id !== page); }); document.querySelectorAll("[data-page]").forEach(button => { button.classList.toggle("active", button.dataset.page === page); }); document.getElementById("heading").textContent = titles[page] || "Central de gestão"; if (page !== "promote") { consultaAtual++; nicknameInput.value = ""; clearPlayerResult(); } } document.querySelectorAll("[data-page]").forEach(button => { button.addEventListener("click", () => openPage(button.dataset.page)); }); document.querySelectorAll("[data-go]").forEach(button => { button.addEventListener("click", () => openPage(button.dataset.go)); }); async function consultarJogador(nickname) { const minhaConsulta = ++consultaAtual; clearPlayerResult(); if (!nickname) return; try { const response = await fetch( `${API_URL}/api/jogador/${encodeURIComponent(nickname)}` ); const data = await response.json(); if (minhaConsulta !== consultaAtual) return; if (!response.ok || !data.encontrado) { errorMessage.textContent = data.mensagem || "Jogador não encontrado."; errorMessage.classList.remove("hidden"); return; } playerName.textContent = data.jogador.nick; playerRank.textContent = data.jogador.cargo; jogadorEncontrado = true; promoteButton.disabled = false; } catch (error) { if (minhaConsulta !== consultaAtual) return; errorMessage.textContent = "Não foi possível conectar à API. Tente novamente."; errorMessage.classList.remove("hidden"); } } let temporizador; nicknameInput.addEventListener("input", () => { clearTimeout(temporizador); consultaAtual++; clearPlayerResult(); const nickname = nicknameInput.value.trim(); if (!nickname) return; temporizador = setTimeout(() => { consultarJogador(nickname); }, 600); }); promoteButton.addEventListener("click", () => { if (!jogadorEncontrado) return; errorMessage.textContent = "Consulta concluída! O sistema de promoção ainda está em desenvolvimento."; errorMessage.classList.remove("hidden"); }); clearPlayerResult();
